01_explore.py (생성 예시 · 핵심부)
# 01_explore.py : 원자료 구조 파악 → codebook.csv
import pyreadstat, pandas as pd

df, meta = pyreadstat.read_sav("data/2022_언론수용자조사.sav")   # 파일명은 실제 이름으로
print("응답자 수:", len(df), "/ 변수 수:", df.shape[1])

rows = []
for col in df.columns:
    rows.append({
        "변수명": col,
        "설명": meta.column_names_to_labels.get(col),
        "값라벨": meta.variable_value_labels.get(col),
        "고유값수": df[col].nunique()
    })
pd.DataFrame(rows).to_csv("codebook.csv", index=False, encoding="utf-8-sig")

res = json.load(open("results.json", encoding="utf-8"))

# 공개 전 점검: 개인 식별 키, 개별 응답 의심 목록
BANNED = {"id", "hid", "pid"}
def check(obj, path="res"):
    if isinstance(obj, dict):
        for k, v in obj.items():
            assert k.lower() not in BANNED, f"식별 키 발견: {path}.{k}"
            check(v, f"{path}.{k}")
    elif isinstance(obj, list):
        assert len(obj) < 2000, f"목록이 너무 김: {path}"
        for i, v in enumerate(obj):
            check(v, f"{path}[{i}]")
check(res)

html = pathlib.Path("template.html").read_text(encoding="utf-8")
data = json.dumps(res, ensure_ascii=False).replace("</", "<\\/")  # 태그 깨짐 방지
pathlib.Path("docs").mkdir(exist_ok=True)
pathlib.Path("docs/index.html").write_text(html.replace("__DATA__", data), encoding="utf-8")
print("docs/index.html 생성:", round(len(data.encode()) / 1024), "KB")

04_api_kosis.py (구조 예시)
import os, requests, pandas as pd
from dotenv import load_dotenv
load_dotenv()                          # .env에서 키 읽기
KEY = os.getenv("KOSIS_KEY")

url = "https://kosis.kr/openapi/Param/statisticsParameterData.do"
params = {
  "method": "getList", "apiKey": KEY,
  "orgId": "기관ID", "tblId": "통계표ID",
  "itmId": "ALL", "objL1": "ALL",
  "prdSe": "Y", "startPrdDe": "2015", "endPrdDe": "2025",
  "format": "json", "jsonVD": "Y"}
data = requests.get(url, params=params, timeout=30).json()
pd.DataFrame(data).to_csv("kosis.csv", index=False, encoding="utf-8-sig")

기관ID·통계표ID는 KOSIS 공유서비스의 "URL 생성" 화면에서 확인. 필요한 파라미터는 통계표마다 상이

GLOSSARY

용어집

본문의 점선 밑줄 용어를 누르면 이곳으로 이동. 오른쪽 아래 "본문으로" 버튼으로 원래 위치 복귀

원자료(원시자료)
조사 문항과 응답자 개개인의 실제 응답을 담은 기초 데이터. 보고서의 표는 이것을 집계한 결과
코드북
변수명, 문항 내용, 응답 코드의 의미(1=예, 2=아니오, 9=모름)를 정리한 사전
결측값·결측 코드
응답이 없거나 '모름·무응답'인 값. 9, 99 등 숫자로 기록돼 있어 그대로 계산하면 평균 왜곡
가중치
표본의 성·연령·지역 구성을 모집단 구성에 맞추기 위해 응답자마다 부여한 보정값. 가중치를 곱해 계산해야 보고서 수치 재현
표본오차
전체 대신 일부만 조사해서 생기는 추정의 불확실성. ±0.4%p는 95% 신뢰수준에서의 오차 범위
신뢰구간(CI)
모집단 값이 있을 것으로 추정되는 범위. 95% CI [9.4, 10.0]처럼 표기
군집(가구 단위 표본)
같은 가구 구성원처럼 서로 비슷한 응답자 묶음. 독립 응답으로 취급하면 표준오차 과소추정
대표본 함정
표본이 매우 크면 의미 없을 만큼 작은 차이도 통계적으로 유의하게 나오는 현상. 효과크기로 판단
리커트 척도
"전혀 그렇지 않다(1)~매우 그렇다(5)"처럼 동의 정도를 묻는 척도. 서열 자료지만 평균으로 요약하는 경우 많음
다중응답
여러 보기를 동시에 고를 수 있는 문항. 보기마다 0/1 변수로 저장
재범주화
연속 연령을 연령대로, 5점 정치성향을 3범주로 묶는 등 값을 다시 분류하는 작업
역코딩
질문 방향이 반대인 문항의 점수를 뒤집는 작업(1↔5). 지수 합산 전 필수 확인
카이제곱 검정
두 범주형 변수(연령대 × 열독 여부)가 서로 관련 있는지 검정
Cramér's V
교차분석의 효과크기. 0~1, 대략 .1 작음 · .3 중간 · .5 큼
t검정(Welch)
두 집단 평균 차이 검정. Welch 방식은 두 집단의 분산이 달라도 사용 가능
Cohen's d
두 평균 차이를 표준편차 단위로 나타낸 효과크기. 대략 .2 작음 · .5 중간 · .8 큼
분산분석(ANOVA)
세 집단 이상의 평균 차이 검정. 순위 기반 대안은 Kruskal-Wallis
사후검정
분산분석 후 어느 집단 쌍이 다른지 확인하는 검정(Games-Howell 등)
η²(에타제곱)
분산분석의 효과크기. 집단 구분이 설명하는 분산 비율. .01 작음 · .06 중간 · .14 큼
대응표본 비교
같은 사람이 두 번 답한 값(TV 신뢰 vs 동영상 신뢰)을 비교. 대응 t검정, Wilcoxon 검정
상관계수
두 변수가 함께 움직이는 정도(-1~1). 리커트·비정규 자료는 Spearman 사용
Cronbach α
여러 문항이 같은 개념을 일관되게 재는지 보는 신뢰도 지표. 보통 .7 이상이면 지수로 묶음
회귀분석(OLS)
여러 변수를 함께 넣어 결과 변수와의 관련성을 각각 추정. 다른 변수를 통제한 뒤의 관계 확인
로지스틱 회귀
결과가 0/1(열독 여부)일 때 쓰는 회귀. 결과는 오즈비로 해석
오즈비(OR)
한 집단의 발생 가능성이 기준 집단의 몇 배인지 나타내는 값. 1보다 크면 높음
k-means 군집분석
비슷한 응답 패턴을 가진 사람끼리 k개 집단으로 묶는 방법. 뉴스 이용 유형 도출에 활용
p값
차이가 없다고 가정할 때 관찰된 결과 이상이 나올 확률. 차이의 크기나 중요성은 말해주지 않음
효과크기
차이나 관계가 실제로 얼마나 큰지 나타내는 지표(d, V, η², r 등). 대표본에서 해석의 중심
작업 폴더
VS Code에서 연 프로젝트 폴더. AI가 이 안의 파일을 읽고 만들고 실행하는 범위
스크립트
순서대로 실행되는 코드 파일(.py, .js). 다시 실행하면 같은 결과 → 재현성의 근거
CLAUDE.md / AGENTS.md
작업 폴더에 두는 지침 파일. AI 코딩 도구가 작업 전에 자동으로 읽는 규칙서
JSON
{"이름": 값} 형태의 텍스트 데이터 형식. Python 결과를 JavaScript로 넘기는 다리
API · API 키
API는 서비스가 데이터를 내주는 공식 창구, API 키는 그 창구용 출입증. 비밀번호처럼 관리
정적 호스팅
HTML·JS·JSON 같은 파일을 그대로 보내주기만 하는 웹 호스팅. 서버에서 Python 실행 불가
GitHub Pages
GitHub 저장소의 파일을 무료 웹사이트로 공개하는 기능. 주소: 아이디.github.io/저장소
Pyodide
브라우저 안에서 Python을 실행하게 해주는 도구. 서버 없이 사용자 컴퓨터에서 계산
저장소(Repository)
GitHub에 올린 프로젝트 폴더와 그 변경 이력 전체
커밋
변경 사항을 메모와 함께 저장한 지점. 언제든 이전 상태로 되돌리기 가능
푸시·동기화
내 컴퓨터의 커밋을 GitHub 저장소로 올리는 작업
.gitignore
GitHub에 올리지 않을 파일·폴더 목록. 원자료·API 키 보호의 첫 장치
스킬(SKILL.md)
반복 작업의 절차·규칙·스크립트를 한 폴더로 묶어 AI가 필요할 때 불러 쓰게 한 재사용 단위
표준편차
값들이 평균에서 평균적으로 얼마나 떨어져 있는지 나타내는 흩어짐 지표
이원분산분석·상호작용
두 집단 변수(성별, 정치성향)의 효과를 함께 보는 분산분석. 상호작용은 한 변수의 효과가 다른 변수 수준에 따라 달라지는 현상
β(표준화 계수)·R²
β는 변수 단위를 표준편차로 맞춘 회귀계수로 변수 간 관련 크기 비교에 사용. R²는 모형이 설명하는 결과 변수 분산의 비율
요인분석(EFA/CFA)
여러 문항 뒤에 있는 공통 잠재요인을 찾는 방법. EFA는 구조 탐색, CFA는 가정한 구조가 데이터에 맞는지 확인
구조방정식모형(SEM)·경로분석
잠재변수 측정과 변수 간 경로 가설을 한 모형으로 동시에 추정. 적합도(CFI, RMSEA)로 모형 평가
매개·조절효과
매개는 X가 M을 거쳐 Y에 이르는 간접 경로, 조절은 X와 Y의 관계 강도가 W에 따라 달라지는 현상
PROCESS
Hayes가 만든 매개·조절 분석 도구와 모델 번호 체계(모델 4 매개, 모델 1 조절, 모델 7 조절된 매개). Python에서는 회귀와 부트스트랩으로 구현
부트스트랩
표본에서 반복 재추출해 추정값의 분포와 신뢰구간을 구하는 방법. 간접효과 검정의 표준
네트워크 분석·중심성·커뮤니티
노드와 연결로 관계 구조 분석. 중심성은 노드의 중요도, 커뮤니티는 서로 촘촘히 연결된 하위 집단
계층적 군집
가까운 사례부터 차례로 묶어 나무 구조(덴드로그램)를 만드는 군집 방법
Random Forest
여러 의사결정나무의 예측을 투표로 합치는 머신러닝 모형. 변수 중요도 제공
SVM
두 범주를 가장 넓은 간격으로 가르는 경계를 찾는 분류 모형
데이터 포함 단일 HTML
화면, 차트 코드, 집계 데이터(JSON)를 HTML 파일 하나에 모두 담은 형태. 더블클릭만으로 열리고 파일 1개로 공유·배포
소표본 셀 억제
응답자가 너무 적은 집단(예: 30명 미만)의 값을 공개하지 않는 처리. 불안정한 추정과 개인 식별 위험 방지
CDN
라이브러리 파일을 전 세계 서버에서 빠르게 내려주는 배포망. script 태그 한 줄로 불러오기
